# 构建GAM公式 - 对连续变量使用不同的平滑参数
continuous_vars <- selected_predictors[sapply(train_set[selected_predictors], is.numeric)]
continuous_vars <- continuous_vars[continuous_vars != "Private"] # 排除二值变量

# 为不同变量设置不同的平滑参数
gam_terms <- c()
if("Private" %in% selected_predictors) {
  gam_terms <- c(gam_terms, "Private")
}

for(var in continuous_vars) {
  # 根据变量特性设置不同的平滑参数
  if(var %in% c("Expend", "Room.Board")) {
    gam_terms <- c(gam_terms, paste0("s(", var, ", k=8)")) # 更多自由度
  } else {
    gam_terms <- c(gam_terms, paste0("s(", var, ", k=5)")) # 默认自由度
  }
}

gam_formula <- as.formula(paste("Outstate ~", paste(gam_terms, collapse = " + ")))
cat("GAM模型公式:\n")
## GAM模型公式:
print(gam_formula)
## Outstate ~ Private + s(Accept, k = 5) + s(Top10perc, k = 5) + 
##     s(Room.Board, k = 8) + s(Personal, k = 5) + s(Terminal, k = 5) + 
##     s(S.F.Ratio, k = 5) + s(perc.alumni, k = 5) + s(Expend, k = 8) + 
##     s(Grad.Rate, k = 5)
# 拟合GAM模型
gam_model <- gam(gam_formula, data = train_set, method = "REML")

# 计算多种评估指标
calculate_metrics <- function(actual, predicted) {
  rmse <- sqrt(mean((actual - predicted)^2))
  mae <- mean(abs(actual - predicted))
  mape <- mean(abs((actual - predicted)/actual)) * 100
  rsq <- 1 - sum((actual - predicted)^2) / sum((actual - mean(actual))^2)
  
  return(c(RMSE = rmse, MAE = mae, MAPE = mape, R_squared = rsq))
}

train_metrics <- calculate_metrics(train_set$Outstate, train_pred)
test_metrics <- calculate_metrics(test_set$Outstate, test_pred)

results_df <- data.frame(
  指标 = names(train_metrics),
